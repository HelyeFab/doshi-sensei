// Add these rules to your Firestore security rules (firestore.rules)

// Analytics collection - admin read access, anyone can write events
match /analytics/{analyticsId} {
  allow read: if request.auth != null &&
              request.auth.token.email == "emmanuelfabiani23@gmail.com";
  allow write: if request.auth != null; // Allow any authenticated user to create analytics events
}

// Admin logs - admin only (this should already exist from previous implementation)
match /adminLogs/{logId} {
  allow read, write: if request.auth != null &&
                    request.auth.token.email == "emmanuelfabiani23@gmail.com";
}

// Make sure these exist for mood boards and users
match /moodBoards/{boardId} {
  allow read: if true; // Public read
  allow write: if request.auth != null &&
               request.auth.token.email == "emmanuelfabiani23@gmail.com";
}

match /users/{userId} {
  allow read: if request.auth != null &&
             (request.auth.uid == userId ||
              request.auth.token.email == "emmanuelfabiani23@gmail.com");
  allow write: if request.auth != null &&
              (request.auth.uid == userId ||
               request.auth.token.email == "emmanuelfabiani23@gmail.com");
}
